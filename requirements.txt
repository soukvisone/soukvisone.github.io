#  requirements.txt — รายการเครื่องมือที่สคริปต์และเอกสารในชุดนี้ใช้
#
#  วิธีใช้:
#     bash install.sh              # ติดตั้งเฉพาะ core (พอสำหรับสคริปต์ใน scripts/)
#     bash install.sh --all        # ติดตั้งทุกอย่างรวมของในเอกสาร (กินดิสก์หลาย GB)
#     bash install.sh --dry-run    # ดูว่าจะสั่งอะไรบ้าง ไม่ติดตั้งจริง
#     bash scripts/check_tools.sh  # ตรวจว่ามีอะไรขาด (ไม่ติดตั้ง)
#
#  รูปแบบบรรทัด:  <ตัวจัดการ>  <ชื่อแพ็กเกจ>  <ตัวตรวจ>  <ระดับ>
#     ตัวจัดการ: apt | pip | gem | skip
#     ตัวตรวจ : apt/gem ใช้ชื่อคำสั่ง, pip ใช้ชื่อ module python, skip ใช้ชื่อคำสั่ง
#     ระดับ   : core (จำเป็นกับสคริปต์) | extra (ของเอกสาร/ทางเลือก)
#
#  บรรทัดที่ขึ้นต้นด้วย # = comment (ไม่ถูกอ่าน)
#  หมายเหตุ: ทุกอย่างติดตั้งจาก repo ทางการของ Kali/Ubuntu และ PyPI เท่านั้น
# ==========================================================================
